& $py -u -c @'
import json, math, torch
from pathlib import Path
from nanolm.generate import load_run
from nanolm.models import build_model
from nanolm.device import seed_everything
from nanolm.data import TokenData, byte_lut_tensors
from nanolm.evaluate import evaluate
torch.set_num_threads(4)
root = Path('outputs/nanolm/question1_char_rnn')
for name in ['trained', 'untrained_char', 'untrained_byte']:
    run = root / name
    saved = json.loads((run / 'metrics.json').read_text())
    cfg, model, tok, dinfo, meta = load_run(str(run), device='cpu')
    data = TokenData(meta)
    measured = evaluate(model, data.val_windows(cfg.train.seq_len, cfg.eval.final_eval_tokens), cfg.train.seq_len, cfg.eval.batch_size, byte_lut_tensors(meta, dinfo.device), dinfo)
    assert abs(measured['bpb'] - saved['val_bpb']) < 0.001
    if name != 'trained':
        seed_everything(cfg.seed)
        initial = build_model(cfg.model)
        assert all(torch.equal(v, initial.state_dict()[k]) for k, v in model.state_dict().items())
        print(name, saved['val_loss'], math.log(meta['vocab_size']))
m = json.loads((root / 'trained/metrics.json').read_text())
loss = m['val_loss']; total_nats = loss * m['val_tokens']
ppl = math.exp(loss)
bits_per_token = loss / math.log(2)
bpb = total_nats / (m['val_bytes'] * math.log(2))
assert abs(ppl - m['val_ppl']) < 1e-10
assert abs(bpb - m['val_bpb']) < 1e-10
print({'loss': total_nats/m['val_tokens'], 'ppl': ppl, 'bits_per_token': bits_per_token, 'bpb': bpb})
'@
